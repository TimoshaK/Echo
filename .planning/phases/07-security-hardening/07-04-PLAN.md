```python
"""Static invariant audit for Phase 7 security hardening.

Asserts the cross-file wiring that unit tests cannot see: that the hardening
primitives are actually reachable from the shipped code paths, and that the
patterns Phase 6 deliberately established were not broken.
"""

import pathlib
import sys

# The script lives in %TEMP%, so the repository root comes from the working
# directory. Run it from the repository root (as the task instructs).
ROOT = pathlib.Path.cwd()
if not (ROOT / "echo").is_dir():
    print("FAIL: run this script from the repository root (no echo/ package in %s)" % ROOT)
    sys.exit(1)

failures = []


def read(relative):
    return (ROOT / relative).read_text(encoding="utf-8")


def check(label, condition, detail=""):
    if condition:
        print("PASS  %s" % label)
    else:
        print("FAIL  %s %s" % (label, detail))
        failures.append(label)


llm = read("echo/llm_client.py")
cfg = read("echo/config.py")
err = read("echo/errors.py")
engine = read("echo/summarization_engine.py")
app = read("echo/ui/app.py")
dialog = read("echo/ui/settings_dialog.py")
reqs = read("requirements.txt")

check("SEC-02 validate_base_url defined", "def validate_base_url(url: str) -> str:" in llm)
check("SEC-02 validate_base_url called in post_chat", "validate_base_url(llm.get(" in llm)
check("SEC-02 dialog validates before saving", "validate_base_url(raw)" in dialog)
check("SEC-02 dialog refuses and returns", 'if not ok:' in dialog and "showerror" in dialog)
check("SEC-02 old unvalidated assignment gone", "base_url=url_var.get()" not in dialog)

check("SEC-03 SafeRedirectHandler defined", "class SafeRedirectHandler(" in llm)
check("SEC-03 header popped on cross-origin", 'headers.pop("Authorization", None)' in llm)
check("SEC-03 private opener used", "_http_opener().open(req, timeout=60)" in llm)
check("SEC-03 global opener untouched", "install_opener" not in llm)
check("SEC-03 no bare urlopen", "urllib.request.urlopen(" not in llm)

check("SEC-05 sanitizer defined", "def sanitize_error_detail(" in err)
check("SEC-05 invalid base url type defined", "class InvalidBaseUrlError(ValueError):" in err)
check("SEC-05 used at request layer", llm.count("sanitize_error_detail(") >= 2)
check("SEC-05 used in engine queue payload", "sanitize_error_detail(str(e)" in engine)
check("SEC-05 used before display", "sanitize_error_detail(error, secrets=(api_key,)" in app)
check("T-260921-04 llm_client stays regex-free", "import re" not in llm)
check("T-260921-04 old raw slice removed", "[:500]" not in llm)

check("SEC-01 atomic replace used", "os.replace(tmp_path, path)" in cfg)
check("SEC-01 no in-place truncation", 'open(CONFIG_PATH, "w"' not in cfg)
check("SEC-01 mode constant present", "CONFIG_FILE_MODE = 0o600" in cfg)
check("SEC-01 chmod applied", "os.chmod(tmp_path, mode)" in cfg)
check("SEC-01 windows acl applied", "_restrict_windows_acl(path)" in cfg)
check("REFR-03 config path anchor intact", 'CONFIG_PATH = Path(__file__).resolve().parent.parent / "app_config.json"' in cfg)

check("SEC-04 typed corruption error", "class ConfigCorruptError(RuntimeError):" in cfg)
check("SEC-04 silent json fallback removed", "except (FileNotFoundError, json.JSONDecodeError):" not in cfg)
check("SEC-04 still defaults on missing file", "except FileNotFoundError:" in cfg)
check("SEC-04 engine captures corruption", "except ConfigCorruptError as e:" in engine)
check("SEC-04 app surfaces corruption", "config_error" in app and "_show_config_error" in app)

check("SEC-06 numpy pinned to the verified version", "numpy==2.4.4" in reqs)
check("SEC-06 torch pinned", "torch==2.14.0" in reqs)
check("SEC-06 whisper pinned", "openai-whisper==20250625" in reqs)
check("SEC-06 tqdm pinned", "tqdm==4.70.0" in reqs)
check("SEC-06 no srt declaration", "srt" not in reqs)
check("SEC-06 no range specifiers", ">=" not in reqs)

print("")
if failures:
    print("STATIC_AUDIT_FAILED: %d check(s)" % len(failures))
    sys.exit(1)
print("STATIC_AUDIT_OK")
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Operator's live `app_config.json` → verification procedure | The file holds the real API key and is gitignored; the manual steps deliberately corrupt and overwrite it, so a backup/restore discipline is the control |
| Verification transcript → chat | Anything the operator reports could carry the key if they paste file contents |
| `%TEMP%` audit script → repository | A transient script must not be committed and must not be able to modify source |
