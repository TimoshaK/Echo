```python
"""Persistent application config for pluggable LLM API settings."""

import json
from pathlib import Path

# D-01: app_config.json lives in the REPOSITORY ROOT — one level above the echo/ package.
# Do NOT use Path(__file__).parent here: that resolves to echo/app_config.json, which does
# not exist, and load_config() would silently fall back to DEFAULT_CONFIG (empty API key),
# breaking REFR-03 with no visible error.
CONFIG_PATH = Path(__file__).resolve().parent.parent / "app_config.json"

DEFAULT_CONFIG = {
    "llm": {
        "api_key": "",
        "base_url": "https://openrouter.ai/api/v1",
        "model": "openai/gpt-4o-mini",
        "enabled": False,
        "summary_preset": "free",
    }
}


def load_config() -> dict:
    try:
        with open(CONFIG_PATH, "r", encoding="utf-8") as f:
            cfg = json.load(f)
    except (FileNotFoundError, json.JSONDecodeError):
        return json.loads(json.dumps(DEFAULT_CONFIG))

    for key, value in DEFAULT_CONFIG.items():
        cfg.setdefault(key, value)
    for key, value in DEFAULT_CONFIG["llm"].items():
        cfg["llm"].setdefault(key, value)
    return cfg


def save_config(cfg: dict) -> None:
    normalized = json.loads(json.dumps(DEFAULT_CONFIG))
    normalized["llm"].update(cfg.get("llm", {}))
    with open(CONFIG_PATH, "w", encoding="utf-8") as f:
        json.dump(normalized, f, ensure_ascii=False, indent=2)
```

Notes: `os` is dropped in favour of `pathlib` (D-01 specifies the `Path(...).parent.parent`
form). `open()` accepts a `Path`, so `load_config`/`save_config` bodies are otherwise
unchanged. `app_config.json` itself does NOT move (D-02) — do not touch it.

"""Error contracts: LLM API errors and transcription error message mapping."""


class SummaryApiError(RuntimeError):
    """Ошибка обращения к LLM API.

    `code` — HTTP-статус, 0 — сетевая ошибка, None — некорректный/пустой
    ответ (провал слоя). Наследуемся от RuntimeError осознанно: существующие
    `_run_summary`/UI ловят `Exception`, а `summarize` умеет пробрасывать
    терминальные ошибки без потери кода (`_is_layer_failure`).
    """

    def __init__(self, message: str, code: int | None = None) -> None:
        super().__init__(message)
        self.code = code


def map_transcription_error(error: str) -> str:
    """Map a raw transcription error string to a user-facing Russian message.

    Extracted verbatim from `TranscriberApp._handle_transcription_error` so the
    mapping stays a pure function. The status-widget updates and the messagebox
    stay in `echo/ui/app.py`.

    Branch ORDER is significant and must not be reordered: `ffmpeg` is tested
    before `format`, and the `model` branch is only reached when `format`/`codec`
    are absent.
    """
    error_lower = error.lower()

    if "ffmpeg" in error_lower or "avconv" in error_lower:
        return "Не установлен ffmpeg. Установите ffmpeg и попробуйте снова."
    elif "format" in error_lower or "codec" in error_lower:
        return "Неподдерживаемый формат файла. Используйте MP3, WAV, M4A, FLAC, OGG или WebM."
    elif "model" in error_lower or "download" in error_lower:
        return "Не удалось загрузить модель Whisper. Проверьте подключение к интернету."
    elif "memory" in error_lower or "allocat" in error_lower:
        return "Недостаточно памяти. Попробуйте файл меньшего размера."
    else:
        return "Ошибка при обработке аудио. Попробуйте другой файл."
```

"""Pure SRT helpers — no tkinter, no file dialogs, no message boxes."""


def time_to_srt(seconds: float, offset: float = 0.0) -> str:
    total = int(seconds + offset)
    millis = int((seconds + offset - total) * 1000)
    hours, rem = divmod(total, 3600)
    minutes, secs = divmod(rem, 60)
    return f"{hours:02d}:{minutes:02d}:{secs:02d},{millis:03d}"


def build_srt_content(segments: list) -> str:
    lines = []
    for idx, seg in enumerate(segments, start=1):
        start = time_to_srt(float(seg.get("start", 0)))
        end = time_to_srt(float(seg.get("end", 0)))
        text = str(seg.get("text", "")).strip()
        lines.append(f"{idx}\n{start} --> {end}\n{text}\n")
    return "\n".join(lines)
```

replaced by the `segments` parameter and `self._time_to_srt` replaced by the module-level
`time_to_srt`. Preserve `.strip()` on the segment text and the trailing `"\n"` per cue.

Do NOT move `save_transcription_txt` / `save_transcription_srt` here — they use

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Local config file → application | `app_config.json` is user-editable and gitignored; its content is read into `llm.*` and later used to build request URLs and headers |
| Filesystem → package import | `CONFIG_PATH` is derived from `__file__`; if it resolves outside the repo root the app silently loses the API key |
| Repo root ↔ `echo/` package | Two config modules coexisting during migration risks split-brain config reads |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01 | I (Information Disclosure) | `echo/config.py` | mitigate | Config values are read into a dict and never formatted into any message or exception text. This plan adds no logging and no error string containing `api_key`; the `Authorization`-header-only contract is enforced in Plan 02 (`echo/llm_client.py`). |
| T-06-02 | T (Tampering) | `summary_preset` read from config | mitigate | Deferred to Plan 02: `echo/summarization_engine.py` must validate the preset against the `SUMMARY_PRESETS` registry and fall back to `DEFAULT_PRESET`. This plan preserves the registry as the single source of truth. |
| T-06-08 | T (Tampering) / D (DoS) | `CONFIG_PATH` anchor | mitigate | Explicit `Path(__file__).resolve().parent.parent / "app_config.json"` form plus a hard assertion that the resolved parent is the repository root and not `echo/`. `load_config()` swallowing `FileNotFoundError` makes this failure silent, so the assertion is the only reliable guard. |
| T-06-09 | D (DoS) | `map_transcription_error` input | accept | Input is an exception string from a local engine, not remote data; the function only does substring checks and returns fixed literals. |
| T-06-10 | I (Information Disclosure) | `echo/__init__.py` eager imports | accept | Package is local code; no secret is reachable from a module import. The no-eager-import rule is a performance/cycle guard, asserted for hygiene. |
</threat_model>
