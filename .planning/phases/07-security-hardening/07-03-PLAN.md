class EngineConfigErrorTests(unittest.TestCase):
    def setUp(self):

        self.path = Path(self._tmpdir.name) / "app_config.json"

    def _write(self, text):
        if isinstance(text, bytes):
            self.path.write_bytes(text)
        else:
            self.path.write_text(text, encoding="utf-8")

    def test_corrupt_config_is_captured_not_raised(self):
        self._write("}{")
        engine = SummarizationEngine()
        self.assertIsNotNone(engine.config_error)
        self.assertIn("app_config.json", engine.config_error)

    def test_corrupt_config_falls_back_to_defaults(self):
        self._write('["not", "an", "object"]')
        engine = SummarizationEngine()
        self.assertEqual(engine.config["llm"]["api_key"], "")
        self.assertFalse(engine.config["llm"]["enabled"])

    def test_corrupt_config_reports_not_configured(self):
        self._write("}{")
        engine = SummarizationEngine()
        self.assertFalse(engine.is_configured())

    def test_corrupt_config_blocks_summarize_with_a_clear_message(self):
        self._write("}{")
        engine = SummarizationEngine()
        with self.assertRaises(RuntimeError) as ctx:
            engine.summarize("какой-то текст")

    def test_missing_config_file_is_not_an_error(self):
        engine = SummarizationEngine()
        self.assertIsNone(engine.config_error)

    def test_valid_config_has_no_config_error(self):

        self.assertTrue(engine.is_configured())

class ValidateSettingsBaseUrlTests(unittest.TestCase):
    def test_live_url_is_accepted(self):
        ok, result = settings_dialog.validate_settings_base_url("https://api.dslab.tech/v1")
        self.assertTrue(ok)
        self.assertEqual(result, "https://api.dslab.tech/v1")

    def test_trailing_slash_is_normalized(self):
        ok, result = settings_dialog.validate_settings_base_url("https://openrouter.ai/api/v1/")
        self.assertTrue(ok)
        self.assertEqual(result, "https://openrouter.ai/api/v1")

    def test_http_is_refused_with_a_message(self):
        ok, message = settings_dialog.validate_settings_base_url("http://example.com/v1")
        self.assertFalse(ok)
        self.assertIn("https://", message)

    def test_empty_is_refused(self):
        ok, message = settings_dialog.validate_settings_base_url("")
        self.assertFalse(ok)

        ok, _ = settings_dialog.validate_settings_base_url(

        """Cancel the startup dialog timer so it cannot fire after teardown.

        except Exception:

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Corrupt config file → application startup | `__init__` runs on the Tk main thread; an uncaught exception here means the window never appears |
| Settings dialog entry → persisted config | The dialog is the only writer of `base_url`, so it is the place where an invalid scheme is cheapest to stop |
| Provider error text → Tk messagebox | The last hop before untrusted remote text is rendered to the user |
| Progress queue → UI thread | The queue payload is an internal boundary, but it is also the value tests and future consumers observe |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-03-01 | R (Repudiation) / I (Information Disclosure) | `SummarizationEngine.__init__` | mitigate | Capture `ConfigCorruptError` into `self.config_error`, fall back to a deep copy of `DEFAULT_CONFIG`, and have `TranscriberApp` show a dialog naming `app_config.json`. `is_configured()` returns False while a corruption is recorded, so a stale in-memory key can never be used. |
| T-07-03-02 | I (Information Disclosure) | `_run_summary` queue payload | mitigate | The payload is sanitized at the source with the configured key as an explicit secret, so a secret cannot leak through any consumer of the queue — not just the messagebox. |
| T-07-03-03 | T (Tampering) / E (Elevation of Privilege) | `settings_dialog.save_settings` | mitigate | `validate_settings_base_url` runs before `update_config`; a failure shows the reason, keeps the dialog open (so the typed key is not lost) and returns without persisting. Combined with `post_chat`'s own validation, a non-https URL can neither be stored nor sent. |
| T-07-03-04 | I (Information Disclosure) | `_handle_summary_error` | mitigate | Second sanitization barrier before display, with `limit=500` and the configured key passed explicitly; an empty sanitized result renders a placeholder instead of a bare dialog.
