```python
"""Mechanical UI-construction parity check for the echo refactor (REFR-02).

Extracts every tkinter widget/variable constructor call from the baseline
`main.py` (via `git show <baseline>:main.py`) and from the refactored UI modules,
normalizes them, and asserts the multisets are identical.

Usage:
    py -3 echo_widget_parity.py <baseline_commit>

Exit code 0 and "WIDGET_PARITY_EXACT" on success; non-zero with a diff otherwise.
"""

import collections
import re
import subprocess
import sys

WIDGETS = (
    "Frame", "Label", "Button", "Entry", "Text", "Scrollbar", "Toplevel",
    "Checkbutton", "StringVar", "BooleanVar", "IntVar", "Canvas", "Listbox",
    "Menu", "Radiobutton", "Scale", "Spinbox", "Combobox", "Style", "Notebook",
    "PanedWindow", "LabelFrame",
)
CALL_RE = re.compile(r"\b(?:tk|ttk)\.([A-Z][A-Za-z]*)\(")
NEW_FILES = [
    "echo/ui/build.py",
    "echo/ui/settings_dialog.py",
    "echo/ui/app.py",
]


def extract(source: str) -> list:
    """Return normalized constructor calls (full argument text) in source order."""
    found = []
    for match in CALL_RE.finditer(source):
        if match.group(1) not in WIDGETS:
            continue
        depth = 0
        i = match.end() - 1  # points at the opening '('
        start = match.start()
        while i < len(source):
            ch = source[i]
            if ch in "([{":
                depth += 1
            elif ch in ")]}":
                depth -= 1
                if depth == 0:
                    break
            elif ch in "\"'":
                quote = ch
                i += 1
                while i < len(source) and source[i] != quote:
                    if source[i] == "\\":
                        i += 1
                    i += 1
            i += 1
        raw = source[start:i + 1]
        norm = re.sub(r"\s+", " ", raw)
        norm = norm.replace("self.", "app.")
        norm = re.sub(r",\s*\)", ")", norm)
        norm = re.sub(r"\(\s+", "(", norm)
        found.append(norm)
    return found


def main() -> int:
    baseline_commit = sys.argv[1] if len(sys.argv) > 1 else "6887892"
    baseline_src = subprocess.run(
        ["git", "show", f"{baseline_commit}:main.py"],
        capture_output=True, text=True, encoding="utf-8", check=True,
    ).stdout

    old = extract(baseline_src)
    new = []
    for path in NEW_FILES:
        with open(path, "r", encoding="utf-8") as fh:
            new.extend(extract(fh.read()))

    old_c, new_c = collections.Counter(old), collections.Counter(new)
    missing = old_c - new_c
    extra = new_c - old_c

    print(f"BASELINE_CONSTRUCTORS={len(old)} REFACTORED_CONSTRUCTORS={len(new)}")
    if missing or extra:
        print("--- MISSING IN REFACTOR (present in baseline) ---")
        for item, n in missing.items():
            print(f"  x{n}: {item[:300]}")
        print("--- EXTRA IN REFACTOR (absent from baseline) ---")
        for item, n in extra.items():
            print(f"  x{n}: {item[:300]}")
        print("WIDGET_PARITY_MISMATCH")
        return 1

    print("WIDGET_PARITY_EXACT")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
